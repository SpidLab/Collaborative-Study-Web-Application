# ============================================================
# Test Scenarios for Orchestration Server
# ============================================================

# 1. Single Request (Should start immediately)

curl -X POST http://localhost:8080/orchestrate \
  -H "Authorization: Bearer user-token-12345" \
  -H "Content-Type: application/json" \
  -d '{
    "metadata": [
      {"key": "userId", "value": "user123"},
      {"key": "taskType", "value": "data-processing"},
      {"key": "priority", "value": "high"}
    ]
  }'

# 2. Multiple Requests (First 5 start, rest queue)
for i in {1..8}; do
  curl -X POST http://localhost:8080/orchestrate \
    -H "Authorization: Bearer user-token-$i" \
    -H "Content-Type: application/json" \
    -d "{
      \"metadata\": [
        {\"key\": \"userId\", \"value\": \"user$i\"},
        {\"key\": \"taskType\", \"value\": \"compute\"},
        {\"key\": \"iteration\", \"value\": \"$i\"}
      ]
    }"
  sleep 0.5
done

# 3. Check Status
curl -X GET http://localhost:8080/status

# 4. Test Different Metadata Schemas
curl -X POST http://localhost:8080/orchestrate \
  -H "Authorization: Bearer test-token" \
  -H "Content-Type: application/json" \
  -d '{
    "metadata": [
      {"key": "dataSource", "value": "database-1"},
      {"key": "outputFormat", "value": "json"},
      {"key": "compression", "value": "gzip"}
    ]
  }'

# 5. Invalid Request (Missing metadata)
curl -X POST http://localhost:8080/orchestrate \
  -H "Authorization: Bearer test-token" \
  -H "Content-Type: application/json" \
  -d '{}'

# 6. Unauthorized Request
curl -X POST http://localhost:8080/orchestrate \
  -H "Content-Type: application/json" \
  -d '{"metadata": [{"key": "test", "value": "data"}]}'
